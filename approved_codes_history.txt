# Calculate combinations: 12 choose 4
# Formula: n! / (k! * (n-k)!)
combinations = math.factorial(12) // (math.factorial(4) * math.factorial(12 - 4))

# Calculate GCD of 252 and 105
gcd_value = math.gcd(252, 105)

# Calculate the final result
result = combinations + gcd_value

print(result)
